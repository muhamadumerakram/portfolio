Helper — Random string generator

export const generateRandomString = (length = 11) => {
  const randomNumber = Math.random()
  const randomString = randomNumber.toString(36).substring(2)

  if (randomString.length < length) {
    return randomString.padEnd(length, '0')
  }
  return randomString.slice(0, length)
}

Page object — Login page

class LoginPage {
  get emailField()    { return cy.get('[data-cy="email-input"]') }
  get passwordField() { return cy.get('[data-cy="password-input"]') }
  get submitBtn()     { return cy.get('[data-cy="login-btn"]') }
  get errorMsg()      { return cy.get('[data-cy="error-message"]') }

  visit() {
    cy.visit('/login')
    return this
  }

  login(email, password) {
    this.emailField.clear().type(email)
    this.passwordField.clear().type(password)
    this.submitBtn.click()
    return this
  }

  assertError(message) {
    this.errorMsg.should('be.visible').and('contain', message)
    return this
  }
}
export default LoginPage

Test execution — Login suite

import LoginPage from '../../pages/LoginPage'

describe('Login Page Tests', () => {
  const loginPage = new LoginPage()

  beforeEach(() => {
    loginPage.visit()
  })

  it('should login with valid credentials', () => {
    loginPage.login('user@example.com', 'SecurePass123')
    cy.url().should('include', '/dashboard')
  })

  it('should show error for invalid email', () => {
    loginPage.login('invalid-email', 'SecurePass123')
    loginPage.assertError('Please enter a valid email')
  })

  it('should show error for empty password', () => {
    loginPage.login('user@example.com', '{selectall}{backspace}')
    loginPage.assertError('Password is required')
  })
})

Parameterized tests — Input validation

#include <gtest/gtest.h>
#include "validator.h"

struct EmailTestCase {
    std::string email;
    bool expected;
    std::string description;
};

class EmailValidatorTest
    : public ::testing::TestWithParam<EmailTestCase> {};

TEST_P(EmailValidatorTest, ValidatesEmailFormat) {
    const auto& tc = GetParam();
    EXPECT_EQ(Validator::isValidEmail(tc.email), tc.expected)
        << "Failed for: " << tc.description;
}

INSTANTIATE_TEST_SUITE_P(
    EmailTests,
    EmailValidatorTest,
    ::testing::Values(

        EmailTestCase{"no-at-sign.com",   false, "missing @ symbol"},

);

Mock objects — Service layer testing

#include <gtest/gtest.h>
#include <gmock/gmock.h>
#include "database.h"
#include "user_service.h"

using ::testing::Return;
using ::testing::_;

class MockDatabase : public IDatabase {
public:
    MOCK_METHOD(bool, connect, (), (override));
    MOCK_METHOD(User, findUser, (const std::string& id), (override));
    MOCK_METHOD(bool, saveUser, (const User& user), (override));
};

class UserServiceTest : public ::testing::Test {
protected:
    MockDatabase mockDb;
    UserService service{&mockDb};
};

TEST_F(UserServiceTest, FindsExistingUser) {
    User expected{"123", "John Doe", "john@test.com"};
    EXPECT_CALL(mockDb, findUser("123"))
        .WillOnce(Return(expected));

    auto result = service.getUserById("123");
    EXPECT_EQ(result.name, "John Doe");
    EXPECT_EQ(result.email, "john@test.com");
}

TEST_F(UserServiceTest, SaveUserCallsDatabase) {
    User newUser{"456", "Jane Smith", "jane@test.com"};
    EXPECT_CALL(mockDb, saveUser(_))
        .WillOnce(Return(true));

    EXPECT_TRUE(service.createUser(newUser));
}

#include <gtest/gtest.h>
#include "string_utils.h"

TEST(StringUtilsTest, TrimsWhitespace) {
    EXPECT_EQ(StringUtils::trim("  hello  "), "hello");
    EXPECT_EQ(StringUtils::trim("\t\n"), "");
    EXPECT_EQ(StringUtils::trim("no-trim"), "no-trim");
}

TEST(StringUtilsTest, SplitsStringByDelimiter) {
    auto parts = StringUtils::split("a,b,c", ',');
    ASSERT_EQ(parts.size(), 3u);
    EXPECT_EQ(parts[0], "a");
    EXPECT_EQ(parts[1], "b");
    EXPECT_EQ(parts[2], "c");
}

TEST(StringUtilsTest, ConvertsToUpperCase) {
    EXPECT_EQ(StringUtils::toUpper("hello"), "HELLO");
    EXPECT_EQ(StringUtils::toUpper("Hello World"), "HELLO WORLD");
    EXPECT_EQ(StringUtils::toUpper(""), "");
}
